```dart
// lib/features/logistics/domain/usecases/create_pick_up_usecase.dart
import 'package:mdmpi_mobile_app/base/utils/result.dart';
import 'package:mdmpi_mobile_app/features/logistics/domain/entities/pick_up_entity.dart';
import 'package:mdmpi_mobile_app/features/logistics/domain/repositories/i_pick_up_repository.dart';
import 'package:mdmpi_mobile_app/data/repositories/image/i_image_repository.dart';

/// Use case for creating a new pick-up request.
/// 
/// Responsibilities:
/// - Validate input data
/// - Create pick-up entity
/// - Persist to repository
/// - Upload associated images and signature
/// 
/// Returns [Result<PickUpEntity>] indicating success or failure.
class CreatePickUpUseCase {
  final IPickUpRepository _pickUpRepository;
  final IImageRepository _imageRepository;
  
  CreatePickUpUseCase({
    required IPickUpRepository pickUpRepository,
    required IImageRepository imageRepository,
  })  : _pickUpRepository = pickUpRepository,
        _imageRepository = imageRepository;
  
  /// Executes the use case with the provided request data.
  Future<Result<PickUpEntity>> execute(CreatePickUpRequest request) async {
    // 1. Validation (clear and centralized)
    final validation = _validate(request);
    if (validation.isFailure) {
      return Result.failure(validation.error);
    }
    
    // 2. Create entity
    final entity = PickUpEntity(
      id: '', // Generated by backend
      client: request.client,
      itemCategory: request.itemCategory,
      preparedBy: request.preparedBy,
      itemPreparedAt: request.itemPreparedAt,
      itemPreparedEndAt: request.itemPreparedEndAt,
      datePickUp: request.datePickUp,
      remarks: request.remarks,
      status: PickUpStatus.newRequest,
      createdBy: request.createdBy,
      createdAt: DateTime.now(),
      updatedAt: DateTime.now(),
    );
    
    // 3. Persist
    final result = await _pickUpRepository.create(entity);
    if (result.isFailure) {
      return Result.failure(result.error);
    }
    
    final createdEntity = result.value;
    
    // 4. Upload images
    if (request.images.isNotEmpty) {
      final imageResult = await _imageRepository.uploadMultiple(
        requestId: createdEntity.id,
        requestType: 'PickUp',
        images: request.images,
      );
      if (imageResult.isFailure) {
        // Log but don't fail the entire operation
        logDebug('Failed to upload images: ${imageResult.error}');
      }
    }
    
    // 5. Upload signature
    if (request.signature != null) {
      final signatureResult = await _imageRepository.uploadSignature(
        requestId: createdEntity.id,
        signature: request.signature!,
      );
      if (signatureResult.isFailure) {
        logDebug('Failed to upload signature: ${signatureResult.error}');
      }
    }
    
    return Result.success(createdEntity);
  }
  
  /// Validates the create request data.
  Result<void> _validate(CreatePickUpRequest request) {
    if (request.client.id.isEmpty) {
      return Result.failure('Client is required');
    }
    
    if (request.itemCategory.id.isEmpty) {
      return Result.failure('Item category is required');
    }
    
    if (request.preparedBy.isEmpty) {
      return Result.failure('Prepared by field is required');
    }
    
    if (request.datePickUp.isBefore(DateTime.now())) {
      return Result.failure('Pick-up date cannot be in the past');
    }
    
    return Result.success(null);
  }
}
```

```dart
  /// Updates an existing pick-up request.

  /// Returns [Result<PickUpEntity>] with the updated pick-up.
```

```dart
// lib/data/repositories/pick_up/pick_up_repository.dart
import 'package:mdmpi_mobile_app/features/logistics/domain/repositories/i_pick_up_repository.dart';
import 'package:mdmpi_mobile_app/features/logistics/domain/entities/pick_up_entity.dart';
import 'package:mdmpi_mobile_app/data/models/pick_up_model.dart';
import 'package:mdmpi_mobile_app/base/utils/result.dart';

/// Concrete implementation of [IPickUpRepository].
/// 
/// Handles API communication and local database synchronization for pick-up requests.
class PickUpRepository implements IPickUpRepository {
  final PickUpDao _dao;
  final http.Client _httpClient;
  final NetworkManager _networkManager;
  
  PickUpRepository({
    required PickUpDao dao,
    required http.Client httpClient,
    required NetworkManager networkManager,
  })  : _dao = dao,
        _httpClient = httpClient,
        _networkManager = networkManager;
  
  @override
  Future<Result<List<PickUpEntity>>> getAll({bool forceRefresh = false}) async {
    try {
      final isConnected = await _networkManager.isConnected();
      
      // Try local database first if not forcing refresh
      if (!forceRefresh) {
        final localModels = await _dao.getAllPickUps();
        if (localModels.isNotEmpty) {
          final entities = localModels.map((m) => m.toEntity()).toList();
          return Result.success(entities);
        }
      }
      
      // Fetch from API if connected
      if (!isConnected) {
        return Result.failure('No internet connection');
      }
      
      final response = await _httpClient.get(_uri('/api4/RequestPickUp'));
      
      if (response.statusCode == 200) {
        final json = jsonDecode(response.body);
        final List<dynamic> list = _decodeRootToList(json);
        
        final models = list.map((item) => PickUpModel.fromJson(item)).toList();
        
        // Cache locally
        await _dao.insertAll(models);
        
        // Convert to entities
        final entities = models.map((m) => m.toEntity()).toList();
        return Result.success(entities);
      } else {
        return Result.failure('Failed to fetch pick-ups: ${response.statusCode}');
      }
    } catch (e) {
      logDebug('Error in PickUpRepository.getAll: $e');
      return Result.failure('An error occurred: ${e.toString()}');
    }
  }
  
  @override
  Future<Result<PickUpEntity>> create(PickUpEntity entity) async {
    try {
      // Convert entity to DTO/Model
      final model = PickUpModel.fromEntity(entity);
      
      final isConnected = await _networkManager.isConnected();
      if (!isConnected) {
        return Result.failure('No internet connection');
      }
      
      final response = await _httpClient.post(
        _uri('/api4/RequestPickUp'),
        headers: {'Content-Type': 'application/json'},
        body: jsonEncode(model.toJson()),
      );
      
      if (response.statusCode == 200 || response.statusCode == 201) {
        final json = jsonDecode(response.body);
        final createdModel = PickUpModel.fromJson(json);
        
        // Save to local database
        await _dao.insert(createdModel);
        
        return Result.success(createdModel.toEntity());
      } else {
        return Result.failure('Failed to create pick-up: ${response.statusCode}');
      }
    } catch (e) {
      logDebug('Error in PickUpRepository.create: $e');
      return Result.failure('An error occurred: ${e.toString()}');
    }
  }
  
  // ... other methods
}
```

```dart
/// Pure domain entity for pick-up requests.
/// 
/// Contains only business logic and domain concepts.
/// No serialization, no framework dependencies.
class PickUpEntity {
  final String id;
  final ClientEntity client;
  final ItemCategoryEntity itemCategory;
  final String preparedBy;
  final DateTime itemPreparedAt;
  final DateTime itemPreparedEndAt;
  final DateTime datePickUp;
  final String remarks;
  final PickUpStatus status;
  final String? releasedBy;
  final String? receivedBy;
  final String createdBy;
  final DateTime createdAt;
  final DateTime updatedAt;
  final List<String> documentReferences;
  
  PickUpEntity({
    required this.id,
    required this.client,
    required this.itemCategory,
    required this.preparedBy,
    required this.itemPreparedAt,
    required this.itemPreparedEndAt,
    required this.datePickUp,
    required this.remarks,
    required this.status,
    this.releasedBy,
    this.receivedBy,
    required this.createdBy,
    required this.createdAt,
    required this.updatedAt,
    this.documentReferences = const [],
  });
  
  // ========================================================================
  // DOMAIN METHODS (Business Logic)
  // ========================================================================
  
  /// Checks if this pick-up request can be cancelled.
  bool canBeCancelled() {
    return status == PickUpStatus.newRequest || status == PickUpStatus.forPickUp;
  }
  
  /// Checks if the pick-up date is overdue.
  bool isOverdue() {
    return DateTime.now().isAfter(datePickUp) && !isCompleted();
  }
  
  /// Checks if the pick-up is completed.
  bool isCompleted() {
    return status == PickUpStatus.received;
  }
  
  /// Checks if the pick-up is in a terminal state (completed or cancelled).
  bool isTerminal() {
    return status == PickUpStatus.received || status == PickUpStatus.cancelled;
  }
  
  /// Checks if the current user can edit this pick-up.
  bool canBeEditedBy(String userInitial) {
    return createdBy == userInitial && status == PickUpStatus.newRequest;
  }
  
  /// Returns the next valid status transition options.
  List<PickUpStatus> getNextValidStatuses() {
    switch (status) {
      case PickUpStatus.newRequest:
        return [PickUpStatus.forPickUp, PickUpStatus.cancelled];
      case PickUpStatus.forPickUp:
        return [PickUpStatus.pickedUp, PickUpStatus.cancelled];
      case PickUpStatus.pickedUp:
        return [PickUpStatus.received];
      case PickUpStatus.received:
      case PickUpStatus.cancelled:
        return []; // Terminal states
    }
  }
  
  /// Creates a copy with updated fields.
  PickUpEntity copyWith({
    String? id,
    ClientEntity? client,
    ItemCategoryEntity? itemCategory,
    String? preparedBy,
    DateTime? itemPreparedAt,
    DateTime? itemPreparedEndAt,
    DateTime? datePickUp,
    String? remarks,
    PickUpStatus? status,
    String? releasedBy,
    String? receivedBy,
    String? createdBy,
    DateTime? createdAt,
    DateTime? updatedAt,
    List<String>? documentReferences,
  }) {
    return PickUpEntity(
      id: id ?? this.id,
      client: client ?? this.client,
      itemCategory: itemCategory ?? this.itemCategory,
      preparedBy: preparedBy ?? this.preparedBy,
      itemPreparedAt: itemPreparedAt ?? this.itemPreparedAt,
      itemPreparedEndAt: itemPreparedEndAt ?? this.itemPreparedEndAt,
      datePickUp: datePickUp ?? this.datePickUp,
      remarks: remarks ?? this.remarks,
      status: status ?? this.status,
      releasedBy: releasedBy ?? this.releasedBy,
      receivedBy: receivedBy ?? this.receivedBy,
      createdBy: createdBy ?? this.createdBy,
      createdAt: createdAt ?? this.createdAt,
      updatedAt: updatedAt ?? this.updatedAt,
      documentReferences: documentReferences ?? this.documentReferences,
    );
  }
  
  @override
  bool operator ==(Object other) =>
      identical(this, other) ||
      other is PickUpEntity &&
          runtimeType == other.runtimeType &&
          id == other.id;
  
  @override
  int get hashCode => id.hashCode;
}

/// Pick-up request status enumeration.
enum PickUpStatus {
  newRequest,
  forPickUp,
  pickedUp,
  received,
  cancelled;
  
  String get displayName {
    switch (this) {
      case PickUpStatus.newRequest:
        return 'New Request';
      case PickUpStatus.forPickUp:
        return 'For Pick-Up';
      case PickUpStatus.pickedUp:
        return 'Picked Up';
      case PickUpStatus.received:
        return 'Received';
      case PickUpStatus.cancelled:
        return 'Cancelled';
    }
  }
  
  static PickUpStatus fromString(String value) {
    switch (value.toLowerCase()) {
      case 'new request':
        return PickUpStatus.newRequest;
      case 'for pick-up':
        return PickUpStatus.forPickUp;
      case 'picked up':
        return PickUpStatus.pickedUp;
      case 'received':
        return PickUpStatus.received;
      case 'cancelled':
        return PickUpStatus.cancelled;
      default:
        throw ArgumentError('Unknown PickUpStatus: $value');
    }
  }
}
```

```dart
    return PickUpEntity(

      itemCategory: itemCategory?.toEntity() ?? ItemCategoryEntity.empty(),
      preparedBy: preparedBy,
      itemPreparedAt: _parseDateTime(itemPreparedAt),
      itemPreparedEndAt: _parseDateTime(itemPreparedEndAt),
      datePickUp: _parseDateTime(datePickUp),
      remarks: remarks,
      status: PickUpStatus.fromString(status),
      releasedBy: releasedBy.isEmpty ? null : releasedBy,
      receivedBy: receivedBy.isEmpty ? null : receivedBy,
      createdBy: createdBy,
      createdAt: _parseDateTime(createdAt),
      updatedAt: _parseDateTime(updatedAt),
      documentReferences: documentReference,
    );
  }
  
  /// Creates a DTO from a domain entity.
  factory PickUpModel.fromEntity(PickUpEntity entity) {
    return PickUpModel(
      id: entity.id,
      clientId: entity.client.id,
      itemCategoryId: entity.itemCategory.id,
      preparedBy: entity.preparedBy,
      itemPreparedAt: _formatDateTime(entity.itemPreparedAt),
      itemPreparedEndAt: _formatDateTime(entity.itemPreparedEndAt),
      datePickUp: _formatDateTime(entity.datePickUp),
      remarks: entity.remarks,
      status: entity.status.displayName,
      releasedBy: entity.releasedBy ?? '',
      receivedBy: entity.receivedBy ?? '',
      createdBy: entity.createdBy,
      createdAt: _formatDateTime(entity.createdAt),
      updatedAt: _formatDateTime(entity.updatedAt),
      client: ClientModel.fromEntity(entity.client),
      itemCategory: ItemCategoryModel.fromEntity(entity.itemCategory),
      documentReference: entity.documentReferences,
    );

  static DateTime _parseDateTime(String value) {
    if (value.isEmpty) return DateTime.now();
    try {
      return DateTime.parse(value);
    } catch (e) {
      return DateTime.now();
    }
  }
  
  static String _formatDateTime(DateTime dateTime) {
    return dateTime.toIso8601String();
  }
```

### Result Type Implementation

```dart
// lib/base/utils/result.dart

/// A type that represents either a success or a failure.
/// 
/// Use this instead of throwing exceptions for expected error cases.
/// Inspired by Rust's Result<T, E> and Kotlin's Result<T>.
sealed class Result<T> {
  const Result();
  
  /// Creates a successful result with the given value.
  factory Result.success(T value) = Success<T>;
  
  /// Creates a failed result with the given error message.
  factory Result.failure(String error) = Failure<T>;
  
  /// Returns true if this is a successful result.
  bool get isSuccess => this is Success<T>;
  
  /// Returns true if this is a failed result.
  bool get isFailure => this is Failure<T>;
  
  /// Returns the value if successful, otherwise throws.
  T get value {
    if (this is Success<T>) {
      return (this as Success<T>).value;
    }
    throw StateError('Cannot get value from a Failure');
  }
  
  /// Returns the error message if failed, otherwise throws.
  String get error {
    if (this is Failure<T>) {
      return (this as Failure<T>).error;
    }
    throw StateError('Cannot get error from a Success');
  }
  
  /// Transforms the value if successful, otherwise returns the failure.
  Result<U> map<U>(U Function(T value) transform) {
    if (this is Success<T>) {
      return Result.success(transform((this as Success<T>).value));
    }
    return Result.failure((this as Failure<T>).error);
  }
  
  /// Chains another operation if successful, otherwise returns the failure.
  Result<U> flatMap<U>(Result<U> Function(T value) transform) {
    if (this is Success<T>) {
      return transform((this as Success<T>).value);
    }
    return Result.failure((this as Failure<T>).error);
  }
  
  /// Executes one of two callbacks depending on success or failure.
  R fold<R>({
    required R Function(T value) onSuccess,
    required R Function(String error) onFailure,
  }) {
    if (this is Success<T>) {
      return onSuccess((this as Success<T>).value);
    }
    return onFailure((this as Failure<T>).error);
  }
}

/// Successful result containing a value.
class Success<T> extends Result<T> {
  final T value;
  const Success(this.value);
  
  @override
  String toString() => 'Success($value)';
}

/// Failed result containing an error message.
class Failure<T> extends Result<T> {
  final String error;
  const Failure(this.error);
  
  @override
  String toString() => 'Failure($error)';
}
```

## Migration Checklist

### For Each Module (e.g., Pick-Up):

- [ ] **Phase 1: Preparation**
  - [ ] Create `domain/` folder structure
  - [ ] Implement `Result` type if not already done
  - [ ] Update DI bindings

- [ ] **Phase 2: Repository Interface**
  - [ ] Create `domain/repositories/i_<module>_repository.dart`
  - [ ] Update existing repository to implement interface
  - [ ] Update DI binding to use interface
  - [ ] Verify existing code still works

- [ ] **Phase 3: Entities**
  - [ ] Create `domain/entities/<module>_entity.dart`
  - [ ] Add domain methods to entity
  - [ ] Keep existing model as DTO
  - [ ] Add `toEntity()` and `fromEntity()` mappers
  - [ ] Update repository to return entities

- [ ] **Phase 4: Use Cases**
  - [ ] Identify distinct operations (create, update, cancel, etc.)
  - [ ] Create use case class for each operation
  - [ ] Move business logic from DataManager to use cases
  - [ ] Register use cases in DI
  - [ ] Update controller to use use cases
  - [ ] Remove or simplify DataManager

- [ ] **Phase 5: Testing**
  - [ ] Write unit tests for use cases
  - [ ] Write unit tests for repository
  - [ ] Write widget tests for screens
  - [ ] Verify test coverage > 70%

- [ ] **Phase 6: Documentation**
  - [ ] Update module README
  - [ ] Document architecture decisions
  - [ ] Add code examples
